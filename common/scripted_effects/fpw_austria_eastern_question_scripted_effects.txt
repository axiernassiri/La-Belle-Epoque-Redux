### Austria alt-history: Ottoman and Russian wars
### Used by the AUS_rivals_of_the_old / AUS_curtail_russian_ambition branches

### Makes every country on Austria's side of the war make peace with the given country
AUS_side_white_peace_TUR = {
	if = {
		limit = { TUR = { exists = yes } }
		every_country = {
			limit = {
				has_war_with = TUR
				OR = {
					tag = AUS
					is_in_faction_with = AUS
					is_subject_of = AUS
					has_country_flag = AUS_backed_rebel
					tag = PER
					tag = SAU
					tag = GRE
				}
			}
			white_peace = {
				tag = TUR
				message = AUS_ottoman_peace_tt
			}
			set_truce = {
				target = TUR
				days = 365
			}
		}
	}
}

### Gives a rebel country locked militia/line templates and spawns troops in every state it owns
AUS_spawn_rebel_army_BUL = {
	BUL = {
		division_template = {
			name = "Insurgent Militia"
			is_locked = yes
			regiments = {
				militia = { x = 0 y = 0 }
				militia = { x = 0 y = 1 }
				militia = { x = 0 y = 2 }
				militia = { x = 1 y = 0 }
				militia = { x = 1 y = 1 }
				militia = { x = 1 y = 2 }
			}
		}
		division_template = {
			name = "Liberation Infantry"
			is_locked = yes
			regiments = {
				infantry = { x = 0 y = 0 }
				infantry = { x = 0 y = 1 }
				infantry = { x = 0 y = 2 }
				infantry = { x = 1 y = 0 }
				infantry = { x = 1 y = 1 }
				infantry = { x = 1 y = 2 }
			}
		}
		every_owned_state = {
			create_unit = {
				division = "name = \"Insurgent Militia\" division_template = \"Insurgent Militia\" start_experience_factor = 0.05 start_manpower_factor = 1 start_equipment_factor = 1"
				owner = BUL
				count = 2
				allow_spawning_on_enemy_provs = yes
			}
			create_unit = {
				division = "name = \"Liberation Infantry\" division_template = \"Liberation Infantry\" start_experience_factor = 0.1 start_manpower_factor = 1 start_equipment_factor = 1"
				owner = BUL
				count = 1
				allow_spawning_on_enemy_provs = yes
			}
		}
		add_manpower = 20000
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 3000
			producer = AUS
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = 300
			producer = AUS
		}
		add_opinion_modifier = {
			target = AUS
			modifier = medium_increase
		}
	}
}

### Releases a Balkan nation from Ottoman held cores and makes Turkey fight it
AUS_launch_balkan_uprising_BUL = {
	every_state = {
		limit = {
			is_core_of = BUL
			is_owned_by = TUR
		}
		transfer_state_to = BUL
	}
	BUL = {
		set_country_flag = AUS_backed_rebel
	}
	AUS_spawn_rebel_army_BUL = yes
	add_equipment_to_stockpile = {
		type = infantry_equipment
		amount = -3000
		producer = AUS
	}
	TUR = {
		declare_war_on = {
			target = BUL
			type = annex_everything
		}
	}
}
