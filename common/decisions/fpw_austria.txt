AUS_ten_year_settlement_dec = {
	
	AUS_develop_austria = {
		icon = GFX_decision_generic_political_rally
		allowed = { 
			original_tag = AUS
		}
		visible = { 
			
		}
		available = {
			OR = {
				check_variable = {
					austrian_development < hungarian_development
				}
				check_variable = {
					hungarian_development = austrian_development
				}
			}
		} 
		cost = 50
		fire_only_once = no
		days_re_enable = 32
		days_remove = 32
		ai_will_do = {
			factor = 1
		} 
		complete_effect = {
			#AUS_increse_hungarian_attitude_effect = yes
		} 
		remove_effect = {
			random_core_state = {
				limit = {
					NOT = {
						is_core_of = HUN
					}
					NOT = {
						is_core_of = CZE
					}
					NOT = {
						is_core_of = CRO
					}
					NOT = {
						is_core_of = SLV
					}
				}

				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
				add_building_construction = {
					type = arms_factory
					level = 1
					instant_build = yes
				}
				add_to_variable = {
					var = austrian_development
					value = 1 
				}

			}
		}
	}
